bad header: HTTP/1.1 400 Bad Request  [rejected: header without a colon: `not-a-header`]

--- query parsing ---
  "name=ada" -> name = Some(ada)
  "name=" -> name = Some()
  "name" -> name = Some()
  "a=1&name=bob&c=3" -> name = Some(bob)
  "name=a%20b" -> name = Some(a b)
  "name=a+b" -> name = Some(a b)
  "name=%41%42" -> name = Some(AB)
  "name=%zz" -> name = Some(%zz)
  "other=1" -> name = None
  "" -> name = None
--- percent decoding ---
  "plain" -> "plain"
  "a%20b" -> "a b"
  "a+b" -> "a b"
  "%41%42%43" -> "ABC"
  "100%25" -> "100%"
  "%2F" -> "/"
  "%" -> "%"
  "%4" -> "%4"
  "%zz" -> "%zz"
  "caf%C3%A9" -> "café"
  "%E2%82%AC" -> "€"
  "%F0%9F%94%A5" -> "🔥"
  "%C3" -> "�"
  "%C3%28" -> "�("
